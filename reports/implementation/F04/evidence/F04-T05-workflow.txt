name: Infra integration (aws-emulator)

# F04-T05 — CA-05 (P-04 §5). Job de integración contra `aws/aws-emulator`.
# Comandos declarados: los verificados en F00-T03/F00-T04 y F04-T01..T04
# (docker compose con límites G-7, docker cp del runbook §1.3, terraform
# fmt/init/validate/apply, tests/integration/infra/). Sin AWS real, sin
# secretos y sin release/deploy (G-12 no aplica).

# Disparadores: push y pull_request sobre `main`, igual que la CI base F00.
on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

# Permisos mínimos: solo lectura. No se usan secrets ni tokens de escritura.
permissions:
  contents: read

jobs:
  infra-integration:
    name: emulator / terraform apply / tests/integration/infra
    runs-on: ubuntu-latest
    timeout-minutes: 15

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          # Alineado con el entorno local verificado (Node 26.13.0).
          node-version: '26'
          cache: 'npm'

      - name: Setup Terraform
        uses: hashicorp/setup-terraform@v3
        with:
          # Versión fijada, la misma verificada en F00-T04 (AGENTS.md §6).
          terraform_version: '1.17.2'
          # Sin wrapper: `terraform output -json` debe devolver JSON limpio.
          terraform_wrapper: false

      - name: Install
        run: npm ci --no-audit --no-fund

      - name: "Start aws-emulator (G-7: 2 GiB / 2.0 CPUs)"
        run: |
          docker compose config --quiet
          docker compose up -d
          # Límites efectivos exigidos por G-7 (comprobar, no asumir).
          mem="$(docker inspect --format '{{.HostConfig.Memory}}' mediashrink-aws-emulator)"
          cpus="$(docker inspect --format '{{.HostConfig.NanoCpus}}' mediashrink-aws-emulator)"
          echo "memory=${mem} nanoCpus=${cpus}"
          test "${mem}" = "2147483648"
          test "${cpus}" = "2000000000"

      - name: Wait for emulator healthy
        run: |
          for i in $(seq 1 60); do
            status="$(docker inspect --format '{{.State.Health.Status}}' mediashrink-aws-emulator 2>/dev/null || echo starting)"
            if [ "${status}" = "healthy" ]; then
              echo "emulator healthy after ${i} checks"
              exit 0
            fi
            sleep 3
          done
          docker compose logs --tail=80 aws-emulator
          echo "El emulador no alcanzó el estado healthy"
          exit 1

      - name: Export CA and credentials (runbook §1.3)
        run: |
          # El CA y las credenciales se regeneran en CADA arranque (nota N-01):
          # hay que repetir el docker cp incluso tras un arranque limpio en CI.
          mkdir -p .local/aws-emulator
          docker cp mediashrink-aws-emulator:/home/emulator/.aws/emulator/9297/ca.pem .local/aws-emulator/ca.pem
          docker cp mediashrink-aws-emulator:/home/emulator/.aws/emulator/9297/config .local/aws-emulator/config
          docker cp mediashrink-aws-emulator:/home/emulator/.aws/emulator/9297/emulator_credentials .local/aws-emulator/emulator_credentials
          # Healthcheck HTTPS autenticado con el CA del emulador.
          curl --silent --show-error --fail --cacert .local/aws-emulator/ca.pem https://127.0.0.1:9297/__admin/health

      - name: Terraform fmt
        working-directory: infra/terraform
        run: terraform fmt -check

      - name: Terraform init
        working-directory: infra/terraform
        run: terraform init

      - name: Terraform validate
        working-directory: infra/terraform
        run: terraform validate

      - name: Terraform apply (emulator only)
        working-directory: infra/terraform
        run: terraform apply -auto-approve -input=false -var='is_emulator=true'

      - name: Integration tests (infra)
        run: npx vitest run tests/integration/infra

      - name: Stop aws-emulator (always)
        if: always()
        run: |
          docker compose down
          docker compose ps
